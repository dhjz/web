browser-cdn-demo.html — 零构建通过 esm.sh / unpkg CDN 直接引入 (无损直出 + 重编码) 
<!-- 方式一：纯 HTML 零构建环境，直接通过 CDN 引入 @handy-common-utils/media-utils -->
<input type="file" id="videoPicker" accept="video/*" />

<script type="module">
  // 1. 从 esm.sh 或 unpkg CDN 导入 (也可使用 UMD 全局变量 window.HandyMediaUtils)
  import { extractAudio, probeMediaStreams } from 'https://esm.sh/@handy-common-utils/media-utils';

  document.getElementById('videoPicker').addEventListener('change', async (e) => {
    const file = e.target.files[0];
    if (!file) return;

    // 先探测原始视频与音轨元数据
    const meta = await probeMediaStreams(file);
    console.log('原始音轨编码:', meta.audioTracks[0]?.codec);

    // =====================================================================
    // 示例 A：默认原音轨无损直出 (不重编码 / Stream Copy，毫秒级完成)
    // =====================================================================
    const losslessResult = await extractAudio(file, {
      codec: 'copy',          // 保持原音轨编码不重编码 (如 .aac / .opus / .mp3)
      worker: true,           // 启用后台 Worker 避免阻塞主线程
    });
    console.log('无损直出文件:', losslessResult.extension, losslessResult.blob.size);

    // =====================================================================
    // 示例 B：指定目标格式重编码 (转为 192kbps 44.1kHz MP3 或 WAV)
    // =====================================================================
    const mp3Result = await extractAudio(file, {
      format: 'mp3',          // 可选: 'mp3' | 'wav' | 'ogg' | 'm4a' | 'flac'
      sampleRate: 44100,      // 目标采样率 (Hz)
      bitrate: '192k',        // 目标比特率
      worker: true,           // 后台分块异步编码，杜绝页面卡死
      onProgress(ratio) {
        console.log(`重编码进度: ${Math.round(ratio * 100)}%`);
      },
    });

    // 生成下载链接
    const url = URL.createObjectURL(mp3Result.blob);
    console.log('重编码 MP3 下载地址:', url);
  });
</script>
audio-extractor.service.ts — 前端工程化 TypeScript 封装 (无损直出 + 重编码双模式) 
// 安装: npm install @handy-common-utils/media-utils --registry=https://registry.npmmirror.com
import { extractAudio, probeMediaStreams, type ExtractAudioOptions } from '@handy-common-utils/media-utils';

/**
 * 场景 A：原音轨无损提取 (不重编码 / Stream Copy)
 * 适用于快速提取视频原声、保持 100% 原始音质与零 CPU 消耗
 */
export async function extractOriginalAudioLossless(file: File) {
  const { videoTrack, audioTracks } = await probeMediaStreams(file);
  if (!audioTracks.length) throw new Error('视频文件未包含可用音轨');

  const { blob, extension } = await extractAudio(file, {
    codec: 'copy',
    worker: true,
  });

  return {
    blob,
    fileName: `${file.name.replace(/\.[^.]+$/, '')}.${extension}`,
    sourceCodec: audioTracks[0].codec,
    videoResolution: `${videoTrack.width}x${videoTrack.height}`,
  };
}

/**
 * 场景 B：重编码为指定音频格式 (支持 MP3 / WAV / OGG / FLAC，带实时进度回调)
 */
export async function extractAndTranscodeAudio(
  file: File,
  targetFormat: 'mp3' | 'wav' | 'ogg' | 'flac' = 'mp3',
  onProgress?: (percent: number) => void
) {
  const options: ExtractAudioOptions = {
    format: targetFormat,
    sampleRate: 48000,
    channels: 2,
    bitrate: targetFormat === 'mp3' ? '192k' : undefined,
    worker: true,
    onProgress(ratio) {
      onProgress?.(Math.round(ratio * 100));
    },
  };

  const { blob, extension } = await extractAudio(file, options);
  return {
    blob,
    url: URL.createObjectURL(blob),
    fileName: `${file.name.replace(/\.[^.]+$/, '')}.${extension}`,
  };
}
batch-extract.mjs — Node.js 服务端批量剥离原音轨 (Stream Copy) 与批量重编码 
import fs from 'node:fs/promises';
import path from 'node:path';
import { extractAudio, probeMediaStreams } from '@handy-common-utils/media-utils';

/**
 * Node.js 服务端批量处理目录视频 (支持 MP4, MKV, MOV, WebM, AVI, FLV, TS)
 * @param {string} inputDir 输入视频目录
 * @param {string} outputDir 输出音频目录
 * @param {'copy' | 'mp3' | 'wav'} mode 'copy' 为原音轨无损直出，其他为重编码转码
 */
async function batchExtractDir(inputDir, outputDir, mode = 'copy') {
  await fs.mkdir(outputDir, { recursive: true });
  const files = await fs.readdir(inputDir);

  for (const fileName of files) {
    if (!/\.(mp4|mkv|mov|webm|avi|flv|ts|m4v|3gp|ogv)$/i.test(fileName)) continue;
    const inputPath = path.join(inputDir, fileName);

    const { audioTracks } = await probeMediaStreams(inputPath);
    if (!audioTracks.length) continue;

    if (mode === 'copy') {
      // 1. 默认原音轨无损直出 (不重编码 / Stream Copy)
      const res = await extractAudio({
        input: inputPath,
        outputDir,
        codec: 'copy',
      });
      console.log(`[无损直出] ${fileName} -> ${res.outputPath}`);
    } else {
      // 2. 重编码为指定格式 (如 MP3 192kbps 或 WAV 48kHz)
      const res = await extractAudio({
        input: inputPath,
        outputDir,
        format: mode,
        sampleRate: 48000,
        bitrate: mode === 'mp3' ? '192k' : undefined,
      });
      console.log(`[重编码 ${mode.toUpperCase()}] ${fileName} -> ${res.outputPath}`);
    }
  }
}

// 示例调用：先无损直出，或传入 'mp3' 进行统一转码
await batchExtractDir('./videos', './audios-lossless', 'copy');
await batchExtractDir('./videos', './audios-mp3', 'mp3');
