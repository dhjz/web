# 电台流反向代理示例
# 域名：home.199311.xyz:40044
# 浏览器里的地址格式：https://home.199311.xyz:40044/rp/qingting/live/20500054/64k.mp3
# data.js 顶部常量就是 https://home.199311.xyz:40044/rp，所以这里必须开 SSL。
#
# 【放哪里】本文件要 include 到 nginx 的 http {} 里面，和 http 同级放会报错。
#   常见写法：把本文件放到 /etc/nginx/conf.d/ 或 /etc/nginx/sites-enabled/，
#   这些目录的默认 include 已经在 http {} 内，不要再包一层 http {}。
#   例如主配置里应有： http { ... include /etc/nginx/conf.d/*.conf; }
#
# nginx 的 location 路径不能用变量，所以前缀 /rp/ 是固定写在下面的；
# 可变部分（上游主机）统一放在顶部 map 里，新增源只加一行即可。

# ===== 反代常量表 =====
# /rp/<key>/  ->  上游 host（可带端口）
map $uri $rp_upstream {
    default             "";
    ~^/rp/125-62-24-73-8030/    http://125.62.24.73:8030;
    ~^/rp/125-82-171-154-8040/    http://125.82.171.154:8040;
    ~^/rp/183-64-174-171-10124/    http://183.64.174.171:10124;
    ~^/rp/audiolive302-iqilu-com/    http://audiolive302.iqilu.com;
    ~^/rp/cri/    http://sk.cri.cn;
    ~^/rp/l-cztvcloud-com/    http://l.cztvcloud.com;
    ~^/rp/live-hnxttv-com-9601/    http://live.hnxttv.com:9601;
    ~^/rp/live-hznet-tv-1935/    http://live.hznet.tv:1935;
    ~^/rp/live-shaoxing-com-cn/    http://live.shaoxing.com.cn;
    ~^/rp/live-xmcdn-com/    http://live.xmcdn.com;
    ~^/rp/livecdn-pull-cdbs-com-cn/    http://livecdn.pull.cdbs.com.cn;
    ~^/rp/ls-qingting-fm/    http://ls.qingting.fm;
    ~^/rp/lzlive-vojs-cn/    http://lzlive.vojs.cn;
    ~^/rp/pili-live-hls-56-i2863-com/    http://pili-live-hls.56.i2863.com;
    ~^/rp/qingting/    http://lhttp.qingting.fm;
    ~^/rp/radio-chinesemusicworld-com/    http://radio.chinesemusicworld.com;
    ~^/rp/rs-ajmide-com/    http://rs.ajmide.com;
    ~^/rp/satellitepull-cnr-cn/    http://satellitepull.cnr.cn;
    ~^/rp/stream-hndt-com/    http://stream.hndt.com;
    ~^/rp/stream3-hndt-com/    http://stream3.hndt.com;
    ~^/rp/zbbf2-ahbztv-com/    http://zbbf2.ahbztv.com;
}

server {
    listen 40044 ssl;
    server_name home.199311.xyz;

    # proxy_pass 用了变量，nginx 运行期要自己解析上游域名，必须配 resolver
    resolver 223.5.5.5 8.8.8.8 valid=30s;

    # 证书要覆盖 home.199311.xyz，路径按你的实际证书改
    ssl_certificate     /etc/nginx/ssl/home.199311.xyz.crt;
    ssl_certificate_key /etc/nginx/ssl/home.199311.xyz.key;
    ssl_protocols       TLSv1.2 TLSv1.3;
    ssl_session_cache   shared:SSL:10m;
    ssl_session_timeout 10m;

    location /rp/ {
        # 未在上表里登记的 key 直接 404，避免变成开放代理
        if ($rp_upstream = "") { return 404; }

        # CORS 预检（hls.js / fetch 跨域会用到）
        if ($request_method = OPTIONS) {
            add_header Access-Control-Allow-Origin "*" always;
            add_header Access-Control-Allow-Methods "GET, HEAD, OPTIONS" always;
            add_header Access-Control-Allow-Headers "Range, Origin, Content-Type" always;
            add_header Access-Control-Max-Age 86400 always;
            return 204;
        }

        add_header Access-Control-Allow-Origin "*" always;
        add_header Access-Control-Allow-Methods "GET, HEAD, OPTIONS" always;
        add_header Access-Control-Allow-Headers "Range" always;

        # /rp/cnr1/live/x.m3u8 -> /live/x.m3u8
        rewrite ^/rp/[^/]+/(.*)$ /$1 break;

        proxy_pass $rp_upstream;
        proxy_set_header Host $proxy_host;
        proxy_set_header Accept-Encoding "";
        proxy_http_version 1.1;
        proxy_buffering off;
        proxy_connect_timeout 10s;
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
        proxy_redirect off;

        # 若某个 m3u8 里写死了绝对地址，再按对应源取消注释下面两行：
        # sub_filter_types application/vnd.apple.mpegurl application/x-mpegURL text/plain;
        # sub_filter 'http://ngcdn001.cnr.cn/' '/rp/cnr1/';
    }
}
